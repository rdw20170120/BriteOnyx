# Dependencies:
# shellcheck disable=SC2154
source "${boDirLib}"/changeDirectoryIn.bash
source "${boDirLib}"/forbidVariable.bash
source "${boDirLib}"/absolute.bash
source "${boDirLib}"/logging.bash
source "${boDirLib}"/maybeCreateDirectoryTree.bash
source "${boDirLib}"/requireDirectory.bash
source "${boDirLib}"/requireDirectoryIn.bash


forbidVariable prjDirTmp
prjDirTmp="${prjHome}"/.tmp
declare -rx prjDirTmp
logVariable prjDirTmp
maybeCreateDirectoryTree "${prjDirTmp}"

forbidVariable prjDirLib
prjDirLib=${prjHome}/lib
declare -rx prjDirLib
logVariable prjDirLib
requireDirectoryIn prjDirLib

_Script=${prjDirLib}/project_override.bash
require "${_Script}" && source "${_Script}"

changeDirectoryIn prjHome

logDone "Project successfully activated in ${prjHome}"

unset _Script

# TODO: refactor
declare -x boFormatDateGls="%Y%m%d_%H%M%S.%N"


# User portion of PATH
# shellcheck disable=SC2154
boPathExeFramework=${boFramework}/bash/bin
declare -rx boPathExeFramework
boPathExeUserBash=${XDG_CONFIG_HOME}/bash/bin
declare -rx boPathExeUserBash
boPathExeUserGeneral=${HOME}/bin
declare -rx boPathExeUserGeneral
# Order matters!
# Framework-specific executables override user executables
# Bash-specific executables override general executables
boPathUser=${boPathExeFramework}
boPathUser+=":${boPathExeUserBash}"
boPathUser+=":${boPathExeUserGeneral}"
declare -x boPathUser

# Run all "first" scripts to establish the foundation of Bash environment
source "$(require "${boFramework}"/bash/f/all.bash)"

if [[ -t 1 ]]; then
    # Since we have a TTY (terminal) connected, and presumably a user,
    # run all "interaction" scripts
    source "$(require "${boFramework}"/bash/i/all.bash)"
fi

# Run all "last" scripts to finish Bash environment
source "$(require "${boFramework}"/bash/l/all.bash)"
